cmake_minimum_required(VERSION 3.16)
project(piper_drawing)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/SetGripper.srv")

add_executable(draw_executor src/main.cpp src/job.cpp src/planner.cpp src/vendor.cpp src/visualization.cpp)
target_include_directories(draw_executor PRIVATE include)
ament_target_dependencies(draw_executor rclcpp sensor_msgs std_srvs moveit_ros_planning_interface moveit_core
  moveit_msgs tf2_eigen visualization_msgs)
target_link_libraries(draw_executor yaml-cpp)

add_executable(drawing_scene src/scene.cpp)
target_include_directories(drawing_scene PRIVATE include)
ament_target_dependencies(drawing_scene rclcpp visualization_msgs tf2_eigen)
target_link_libraries(drawing_scene yaml-cpp)

add_library(drawing_hardware SHARED src/hardware.cpp)
ament_target_dependencies(drawing_hardware rclcpp sensor_msgs std_srvs hardware_interface pluginlib)
pluginlib_export_plugin_description_file(hardware_interface hardware.xml)

install(TARGETS draw_executor drawing_scene DESTINATION lib/${PROJECT_NAME})
install(TARGETS drawing_hardware LIBRARY DESTINATION lib)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES .setup_assistant DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/piper_bridge DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_service ../../../tests/test_service.cpp
    TIMEOUT 120 ENV ROS_DOMAIN_ID=113 ROS_LOCALHOST_ONLY=1)
  if(TARGET test_service)
    target_include_directories(test_service PRIVATE include)
    ament_target_dependencies(test_service rclcpp moveit_msgs)
  endif()
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
